# Manual boot commands for RK3399
# Execute these commands in U-Boot prompt (=>)

# 1. Initialize MMC
mmc dev 0
mmc rescan

# 2. Load kernel from boot partition (FAT)
fatload mmc 0:2 0x00280000 Image-6.1.115

# 3. Load device tree from boot partition (FAT)
fatload mmc 0:2 0x03000000 rk3399-firefly-aio.dtb

# 4. Set boot arguments (root is partition 4)
setenv bootargs earlycon=uart8250,mmio32,0xff1a0000,1500000n8 root=/dev/mmcblk0p4 rootwait rootfstype=ext4 console=ttyS2,1500000 consoleblank=0 loglevel=7 debug initcall_debug mmc.debug=1 mmc.block=1

# 5. Boot kernel
booti 0x00280000 - 0x03000000
